//2.crear la matriz A
para cada i, j:
    si i == j: A[i][j] = 0
    si hay arista entre i y j: A[i][j] = peso
    si no: A[i][j] = infinito

//3.probar pasar por cada nodo k
para k = 0 hasta n-1:
    para i = 0 hasta n-1:
        para j = 0 hasta n-1:
            A[i][j] = min(A[i][j], A[i][k] + A[k][j])

//4.mostrar resultado
mostrar A
